cmake_minimum_required(VERSION 3.22)
project(false_color_correction VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(ANDROID)
  set(FCC_VULKAN_LIB vulkan)
else()
  find_package(Vulkan REQUIRED)
  set(FCC_VULKAN_LIB Vulkan::Vulkan)
endif()
add_library(false_color_correction STATIC src/Pipeline.cpp)
add_library(fcc::false_color_correction ALIAS false_color_correction)
target_include_directories(false_color_correction PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(false_color_correction PUBLIC ${FCC_VULKAN_LIB})
target_compile_options(false_color_correction PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-O3;-Wall;-Wextra;-Wpedantic;-Werror>)

option(FCC_BUILD_VALIDATION "Build FCC Vulkan validation harness" OFF)
if(FCC_BUILD_VALIDATION)
  add_executable(fcc_validate validation/fcc_validate.cpp)
  target_link_libraries(fcc_validate PRIVATE false_color_correction ${FCC_VULKAN_LIB})
  add_executable(fcc_benchmark validation/fcc_benchmark.cpp)
  target_link_libraries(fcc_benchmark PRIVATE false_color_correction ${FCC_VULKAN_LIB})
endif()
