cmake_minimum_required(VERSION 3.22)
project(vng4_demosaic VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(ANDROID)
  set(VULKAN_LIB vulkan)
else()
  find_package(Vulkan REQUIRED)
  set(VULKAN_LIB Vulkan::Vulkan)
endif()
add_library(vng4 STATIC src/Pipeline.cpp)
add_library(vng4::vng4 ALIAS vng4)
target_include_directories(vng4 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(vng4 PUBLIC ${VULKAN_LIB})
target_compile_options(vng4 PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-O3;-Wall;-Wextra;-Wpedantic;-Werror>)
option(VNG4_BUILD_TESTS "Build Vulkan validation harness" OFF)
option(VNG4_BUILD_BENCHMARK "Build Vulkan benchmark harness" OFF)
if(VNG4_BUILD_TESTS)
  add_executable(vng4_validate validation/vng4_validate.cpp)
  target_link_libraries(vng4_validate PRIVATE vng4 ${VULKAN_LIB})
  target_compile_options(vng4_validate PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic;-Werror>)
endif()

if(VNG4_BUILD_BENCHMARK)
  add_executable(vng4_benchmark validation/vng4_benchmark.cpp)
  target_link_libraries(vng4_benchmark PRIVATE vng4 ${VULKAN_LIB})
  target_compile_options(vng4_benchmark PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic;-Werror>)
endif()
