cmake_minimum_required(VERSION 3.22)
project(video_demosaic_moltenvk LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Vulkan REQUIRED)
add_executable(video_demosaic_moltenvk main.cpp)
target_include_directories(video_demosaic_moltenvk PRIVATE ../../native/vk_common/testing)
target_link_libraries(video_demosaic_moltenvk PRIVATE Vulkan::Vulkan)
target_compile_options(video_demosaic_moltenvk PRIVATE -Wall -Wextra -Wpedantic -Werror)

option(RUNNER_BUILD_FUSED "Build the fused RAW-input tonemap runner" ON)
if(RUNNER_BUILD_FUSED)
set(TONEMAP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../native/tonemap_engine ${CMAKE_CURRENT_BINARY_DIR}/tonemap_engine)
add_executable(video_fused_moltenvk fused.cpp)
target_include_directories(video_fused_moltenvk PRIVATE ../../native/vk_common/testing)
target_link_libraries(video_fused_moltenvk PRIVATE tonemap_engine Vulkan::Vulkan)
target_compile_options(video_fused_moltenvk PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
